@app.route('/api/book-cart', methods=['POST'])
def book_cart():
    data = request.get_json()
    reference_id = data.get('reference_id')
    user_id = data.get('user_id')
    email = data.get('email')
    bookings = data.get('bookings', [])

    try:
        conn = mysql.connector.connect(**tours_db_config)
        cursor = conn.cursor()

        for item in bookings:
            cursor.execute("""
                INSERT INTO cart_bookings (reference_id, user_id, email, title, price, quantity)
                VALUES (%s, %s, %s, %s, %s, %s)
            """, (
                reference_id,
                user_id,
                email,
                item['title'],
                item['price'],
                item['quantity']
            ))

        conn.commit()
        return jsonify({"message": "Cart booking saved successfully."}), 200

    except Exception as e:
        return jsonify({"error": str(e)}), 500

    finally:
        cursor.close()
        conn.close()
